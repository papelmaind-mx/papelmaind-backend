export async function syncPendingSales() {
  const pending = await db.ventas
    .where('estado').anyOf('pendiente_sync', 'conflicto', 'conflicto_corte_cerrado')

    "kimi":     {"input": 0.00007, "output": 0.00014},
    "qwen":     {"input": 0.00010, "output": 0.00020},
    "deepseek": {"input": 0.00014, "output": 0.00028},
    "openai":   {"input": 0.00015, "output": 0.00060},

PRESUPUESTO_MENSUAL_TOKENS = 200_000

class LLMGateway:
    """Motor BYO-LLM con routing inteligente + presupuesto + PII masking."""

    def __init__(self):
        self.providers = {
            "kimi":     KimiClient(),
            "qwen":     QwenClient(),
            "deepseek": DeepSeekClient(),
            "openai":   OpenAIClient(),
            "claude":   ClaudeClient(),
            "local":    LocalModelClient(),  # vLLM/Ollama

    def _mask_pii(self, prompt: str) -> str:
        """Reemplaza wa_id (números) con hash para LFPDPPP."""
        def _hash(match):
            numero = match.group()
            hash_corto = hashlib.sha256(numero.encode()).hexdigest()[:8]
            return f"HASH_{hash_corto}"
        return re.sub(r"\+?52\d{10}", _hash, prompt)

    async def _presupuesto_restante(self, session: AsyncSession, tenant_id: UUID) -> int:
        mes_actual = __import__('datetime').datetime.now(
            __import__('datetime').timezone.utc
        ).strftime("%Y-%m")
        result = await session.execute(
            select(func.coalesce(
                func.sum(TLLMBudgetLog.tokens_input + TLLMBudgetLog.tokens_output), 0
            )).where(
                TLLMBudgetLog.papeleria_id == tenant_id,
                TLLMBudgetLog.mes == mes_actual
            )
        )
        consumidos = int(result.scalar() or 0)
        return max(0, PRESUPUESTO_MENSUAL_TOKENS - consumidos)

    def _elegir_proveedor(self, prompt: str, task_type: TaskType,
                          presupuesto_restante: int) -> str:
        """Routing inteligente según task + contexto + presupuesto."""
        # Regla 1: presupuesto bajo → forzar deepseek
        if presupuesto_restante < PRESUPUESTO_MENSUAL_TOKENS * 0.2:
            return "deepseek"
        # Regla 2: tarea crítica → openai
        if task_type == "tarea_critica":
            return "openai"
        # Regla 3: contexto largo → claude
        if len(prompt) > 32_000:
            return "claude"
        # Regla 4: customer_facing → openai
        if task_type == "customer_facing":
            return "openai"
        # Regla 5: largo_contexto → claude
        if task_type == "largo_contexto":
            return "claude"
        # Default: deepseek (barato)
        return "deepseek"

    async def generate(self, session: AsyncSession, tenant_id: UUID, *,
                       prompt: str, task_type: TaskType = "customer_facing",
                       max_tokens: int = 500, temperature: float = 0.3) -> str:
        """Genera respuesta con routing inteligente + tracking de presupuesto."""
        # 1. PII masking
        prompt_seguro = self._mask_pii(prompt)
        # 2. Presupuesto
        presupuesto = await self._presupuesto_restante(session, tenant_id)
        if presupuesto <= 0:
            raise LLMGatewayError("Presupuesto LLM agotado")
        # 3. Elegir proveedor
        proveedor = self._elegir_proveedor(prompt_seguro, task_type, presupuesto)
        logger.info("llm.routing", proveedor=proveedor, task_type=task_type,
                    presupuesto=presupuesto)
        # 4. Llamar al proveedor
        try:
            respuesta = await self.providers[proveedor].generate(
                prompt_seguro, max_tokens, temperature
            )
            # 5. R3: Registrar uso en TLLMBudgetLog
            tokens_in = len(prompt_seguro) // 4
            tokens_out = len(respuesta) // 4
            costo = Decimal(str(
                (tokens_in / 1000) * COSTOS[proveedor]["input"]
                + (tokens_out / 1000) * COSTOS[proveedor]["output"]
            ))
            session.add(TLLMBudgetLog(
                papeleria_id=tenant_id,
                proveedor=proveedor,
                task_type=task_type,
                tokens_input=tokens_in,
                tokens_output=tokens_out,
                costo_usd=costo,
                mes=__import__('datetime').datetime.now(
                    __import__('datetime').timezone.utc
                ).strftime("%Y-%m"),
            ))
            return respuesta
        except Exception as e:
            logger.error("llm.error", proveedor=proveedor, error=str(e)[:200])
            # Fallback chain
            for fb in ["openai", "deepseek", "local"]:
                if fb == proveedor: continue
                try:
                    return await self.providers[fb].generate(
                        prompt_seguro, max_tokens, temperature
                    )
                except Exception:
                    continue
            raise LLMGatewayError("Todos los proveedores fallaron")

import asyncio
from sqlalchemy import text
from src.infrastructure.db.session import WorkerSessionLocal

async def sweep_reservations_once() -> int:
    """F38: helper canonizado — ejecuta UN ciclo del sweeper.
    Retorna el número de items liberados."""
    async with WorkerSessionLocal() as session:
        result = await session.execute(text("""
            WITH items_expirados AS (
                UPDATE tcarritoitems SET reserva_activa = false
                WHERE reserva_activa AND fecha_expiracion_reserva < NOW()
                RETURNING carrito_id, articulo_id, cantidad
            ), liberacion AS (
                UPDATE tarticulos a
                SET stock_reservado = GREATEST(0, a.stock_reservado - sub.total_cantidad)
                FROM (SELECT articulo_id, SUM(cantidad) total_cantidad
                      FROM items_expirados GROUP BY articulo_id) sub
                WHERE a.id = sub.articulo_id RETURNING a.id
            )
            UPDATE tcarritos c SET estado = 'expirado'
            WHERE c.estado = 'activo'
              AND NOT EXISTS (SELECT 1 FROM tcarritoitems i
                              WHERE i.carrito_id = c.id AND i.reserva_activa)
        """))
        await session.commit()
        return result.rowcount or 0

    while True:

            await sweep_reservations_once()
        except Exception as e:
            print(f"Error Sweeper: {e}")
        await asyncio.sleep(60)

if __name__ == "__main__":
