import hmac, hashlib, os
from fastapi import Request, HTTPException
async def validar_hmac_meta(request: Request) -> None:
    signature = request.headers.get("X-Hub-Signature-256", "").replace("sha256=", "")
    body = await request.body()
    secret = os.getenv("META_APP_SECRET", "")
    if not secret:
        return  # Dev sin secret
    expected = hmac.new(secret.encode(), body, hashlib.sha256).hexdigest()
    if not hmac.compare_digest(signature, expected):
        raise HTTPException(401, "Invalid signature")

async def rag_respond(wa_id, query, ctx, db):
    # Nivel 1: FAQ cache
    cached = await rag_level1(redis, ctx['tenant'], query)
    if cached:
        return cached['respuesta']
    # Nivel 2: vectorial
    query_vec = await embed_query(query)
    sql = text("""
        SELECT contenido_texto, metadata_json
        FROM tiainteraccionesvector
        WHERE papeleria_id = :tid
          AND 1 - (embedding <=> :qv) >= 0.65
        ORDER BY embedding <=> :qv
        LIMIT 5
    """)
    rows = (await db.execute(sql, {"tid": str(ctx['tenant']), "qv": query_vec})).fetchall()

Responde amable y breve en {ctx['lang']}. NO inventes stock ni precios.

# src/application/services/llm_gateway.py
import hashlib, os, re
from typing import Literal
from uuid import UUID
import structlog
from sqlalchemy import func, select, text
from sqlalchemy.ext.asyncio import AsyncSession
from src.infrastructure.db.models import TLLMBudgetLog

logger = structlog.get_logger(__name__)
TaskType = Literal["interno_barato", "customer_facing", "largo_contexto", "tarea_critica"]

COSTOS = {
    "deepseek": {"input": 0.00014, "output": 0.00028},
    "openai": {"input": 0.00015, "output": 0.00060},
    "claude": {"input": 0.00300, "output": 0.01500},
}
PRESUPUESTO_MENSUAL_TOKENS = 200_000

class LLMGatewayError(Exception):
    pass

class LLMGateway:
    def __init__(self):
        self.anthropic_key = os.getenv("ANTHROPIC_API_KEY", "")
        self.openai_key = os.getenv("OPENAI_API_KEY", "")
        self.deepseek_key = os.getenv("DEEPSEEK_API_KEY", "")

    def _enmascarar_pii(self, texto: str) -> str:
        """Reemplaza wa_id (números) con hash para LFPDPPP."""
        def _hash(match):
            numero = match.group()
            hash_corto = hashlib.sha256(numero.encode()).hexdigest()[:8]
            return f"HASH_{hash_corto}"
        return re.sub(r"\+?52\d{10}", _hash, texto)

    async def _presupuesto_restante(self, session: AsyncSession, tenant_id: UUID) -> int:
        mes_actual = __import__('datetime').datetime.now(__import__('datetime').timezone.utc).strftime("%Y-%m")
        result = await session.execute(
            select(func.coalesce(func.sum(TLLMBudgetLog.tokens_input + TLLMBudgetLog.tokens_output), 0))
            .where(TLLMBudgetLog.papeleria_id == tenant_id, TLLMBudgetLog.mes == mes_actual)
        )
        consumidos = int(result.scalar() or 0)
        return max(0, PRESUPUESTO_MENSUAL_TOKENS - consumidos)

    def _elegir_proveedor(self, prompt: str, task_type: TaskType, presupuesto_restante: int) -> str:
        if presupuesto_restante < PRESUPUESTO_MENSUAL_TOKENS * 0.2:
            return "deepseek"
        if task_type == "tarea_critica":
            return "openai"
        if len(prompt) > 32_000:
            return "claude"
        if task_type == "customer_facing":
            return "openai"
        if task_type == "largo_contexto":
            return "claude"
        return "deepseek"

    async def generate(self, session, tenant_id, *, prompt, task_type="customer_facing",
                       max_tokens=500, temperature=0.3) -> str:
        prompt_seguro = self._enmascarar_pii(prompt)
        presupuesto = await self._presupuesto_restante(session, tenant_id)
        if presupuesto <= 0:
            raise LLMGatewayError("Presupuesto LLM agotado")
        proveedor = self._elegir_proveedor(prompt_seguro, task_type, presupuesto)
        logger.info("llm.routing", proveedor=proveedor, task_type=task_type, presupuesto=presupuesto)
        try:
            if proveedor == "deepseek":
                respuesta = await self._llamar_deepseek(prompt_seguro, max_tokens, temperature)
            elif proveedor == "openai":
                respuesta = await self._llamar_openai(prompt_seguro, max_tokens, temperature)
            else:
                respuesta = await self._llamar_claude(prompt_seguro, max_tokens, temperature)
            tokens_in = len(prompt_seguro) // 4
            tokens_out = len(respuesta) // 4
            from decimal import Decimal
            costo = Decimal(str(
                (tokens_in / 1000) * COSTOS[proveedor]["input"]
                + (tokens_out / 1000) * COSTOS[proveedor]["output"]
            ))
            session.add(TLLMBudgetLog(
                papeleria_id=tenant_id, proveedor=proveedor, task_type=task_type,
                tokens_input=tokens_in, tokens_output=tokens_out,
                costo_usd=costo,
                mes=__import__('datetime').datetime.now(__import__('datetime').timezone.utc).strftime("%Y-%m"),
            ))
            return respuesta
        except Exception as e:
            logger.error("llm.error", proveedor=proveedor, error=str(e)[:200])
            for fb in ["openai", "deepseek"]:
                if fb == proveedor: continue
                try:
                    if fb == "openai":
                        return await self._llamar_openai(prompt_seguro, max_tokens, temperature)
                    return await self._llamar_deepseek(prompt_seguro, max_tokens, temperature)
                except Exception:
                    continue
            raise LLMGatewayError("Todos los proveedores fallaron")

    async def _llamar_deepseek(self, prompt, max_tokens, temperature) -> str:
        if not self.deepseek_key:
            return f"[DEEPSEEK SIMULADO] {prompt[:80]}..."
        import httpx
        async with httpx.AsyncClient(timeout=30.0) as client:
            r = await client.post(
                "https://api.deepseek.com/v1/chat/completions",
                headers={"Authorization": f"Bearer {self.deepseek_key}"},
                json={"model": "deepseek-chat", "messages": [{"role": "user", "content": prompt}],
                      "max_tokens": max_tokens, "temperature": temperature},
            )
            r.raise_for_status()
            return r.json()["choices"][0]["message"]["content"]

    async def _llamar_openai(self, prompt, max_tokens, temperature) -> str:
        if not self.openai_key:
            return f"[OPENAI SIMULADO] {prompt[:80]}..."
        import httpx
        async with httpx.AsyncClient(timeout=30.0) as client:
            r = await client.post(
                "https://api.openai.com/v1/chat/completions",
                headers={"Authorization": f"Bearer {self.openai_key}"},
                json={"model": "gpt-4o-mini", "messages": [{"role": "user", "content": prompt}],
                      "max_tokens": max_tokens, "temperature": temperature},
            )
            r.raise_for_status()
            return r.json()["choices"][0]["message"]["content"]

    async def _llamar_claude(self, prompt, max_tokens, temperature) -> str:
        if not self.anthropic_key:
            return f"[CLAUDE SIMULADO] {prompt[:80]}..."
        import httpx
        async with httpx.AsyncClient(timeout=30.0) as client:
            r = await client.post(

                headers={"x-api-key": self.anthropic_key, "anthropic-version": "2023-06-01"},
                json={"model": "claude-3-5-sonnet-20241022", "max_tokens": max_tokens,
                      "temperature": temperature, "messages": [{"role": "user", "content": prompt}]},
            )
            r.raise_for_status()
            return r.json()["content"][0]["text"]

from langgraph.graph import StateGraph, END
from typing import TypedDict, Annotated
import operator

class AgentState(TypedDict):
    messages: Annotated[list, operator.add]
    tenant_id: str
    event: dict
    context: dict
    next_agent: str
    handoff_human: bool
    trace_id: str

def router(state: AgentState) -> str:
    tipo = state['event'].get('eventType', '')
    if tipo == 'whatsapp.inbound.received': return 'concierge'
    if tipo.startswith('appointments'): return 'ventas_citas'
    if tipo == 'inventory.stock.low.detected': return 'supply_chain'
    if tipo == 'sale.completed': return 'marketing'
    if tipo.startswith('payments'): return 'systems_integrator'
    return END

def build_graph():
    try:
        from langgraph.graph import StateGraph, END
        from langgraph.checkpoint.postgres import PostgresSaver
        graph = StateGraph(AgentState)
        graph.add_node('concierge', concierge_agent)
        graph.add_node('ventas_citas', ventas_citas_agent)
        graph.add_node('supply_chain', supply_chain_agent)
        graph.add_node('marketing', marketing_agent)

        graph.set_conditional_entry_point(router)
        for node in ['concierge', 'ventas_citas', 'supply_chain',
                     'marketing', 'systems_integrator']:
            graph.add_edge(node, END)
        checkpointer = PostgresSaver.from_conn_string(WORKER_DATABASE_URL)

async def supply_chain_agent(state: AgentState) -> AgentState:

    payload = event['payload']

    articulo_id = UUID(payload['articuloId'])

        WITH demanda AS (
            SELECT SUM(d.cantidad)/30.0 AS diaria, STDDEV_SAMP(d.cantidad) AS sigma
            FROM tventas v JOIN tdetallesventa d ON d.venta_id = v.id
            WHERE v.papeleria_id = :tid AND d.articulo_id = :aid
              AND v.created_at > NOW() - INTERVAL '30 days'
        )
        SELECT diaria, COALESCE(sigma, 0) AS sigma FROM demanda
    """)
    row = (await db.execute(sql_demanda, {"tid": str(tenant_id), "aid": str(articulo_id)})).first()
    if not row or not row.diaria: return state

        "SELECT MIN(lead_time_dias) FROM tproveedores WHERE papeleria_id = :tid AND activo = true"
    ), {"tid": str(tenant_id)})).scalar() or 7

def _impuestos_totales(detalles) -> tuple[str, str]:
    """F25: Agrega traslados por (Impuesto, TasaOCuota) DESDE las líneas — SAT exige
    agrupar; con IEPS/exento la constante 0.16 generaba XML inválido."""
    grupos: dict[tuple[str, str], list[Decimal]] = {}
    for d in detalles:
        if d.impuesto.tipo == "iva":
            key = ("002", f"{d.impuesto.tasa:.6f}")
            grupos.setdefault(key, [Decimal("0"), Decimal("0")])
            grupos[key][0] += d.importe_base
            grupos[key][1] += d.importe_iva
        elif d.impuesto.tipo == "ieps":
            key = ("003", f"{d.impuesto.tasa:.6f}")
            grupos.setdefault(key, [Decimal("0"), Decimal("0")])
            grupos[key][0] += d.importe_base
            grupos[key][1] += d.importe_ieps
        # exento: sin traslado (ObjetoImp y exención por línea)
    traslados = "".join(
        f'<cfdi:Traslado Base="{base:.2f}" Impuesto="{imp}" '
        f'TipoFactor="Tasa" TasaOCuota="{tasa}" Importe="{importe:.2f}"/>'
        for (imp, tasa), (base, importe) in sorted(grupos.items()))
    total = sum(v[1] for v in grupos.values())
    return traslados, f"{total:.2f}"

import asyncio
from abc import ABC, abstractmethod
from decimal import Decimal
from typing import Optional
from uuid import UUID
import structlog

logger = structlog.get_logger(__name__)

class CourierError(Exception):
    pass

class CourierProvider(ABC):
    @abstractmethod
    async def cotizar(self, origen: dict, destino: dict, peso: float) -> dict: ...
    @abstractmethod
    async def crear_orden(self, envio) -> dict: ...
    @abstractmethod
    async def cancelar(self, tracking_id: str) -> bool: ...

class CourierHub:
    def __init__(self):
        self.providers: dict[str, CourierProvider] = {
            'UBER_DIRECT': UberDirectClient(),
            'DIDI': DidiClient(),
            'INDRIVER': InDriverClient(),
            'MOTOTAXI_SEGURO': MototaxiClient(),
            'RAPPI': RappiClient(),
            'AMAZON_LOGISTICS': AmazonLogisticsClient(),
            'ESTAFETA': EstafetaClient(),
            'DHL_STANDARD': DHLClient(mode='standard'),
            'DHL_EXPRESS': DHLClient(mode='express'),
        }

    async def cotizar_todos(self, session, tenant_id: UUID,
                            origen: dict, destino: dict, peso_kg: Decimal) -> list[dict]:
        """Cotiza en paralelo todos los couriers activos."""
        couriers = (await session.execute(
            select(TCourier).where(TCourier.papeleria_id == tenant_id, TCourier.activo.is_(True))
            .order_by(TCourier.prioridad.asc())
        )).scalars().all()
        tasks = [self._cotizar_uno(session, tenant_id, c, origen, destino, peso_kg) for c in couriers]
        resultados = await asyncio.gather(*tasks, return_exceptions=True)
        cotizaciones = []
        for c, result in zip(couriers, resultados):
            if isinstance(result, Exception):
                session.add(TCotizacionCourier(
                    papeleria_id=tenant_id, courier_codigo=c.codigo,
                    fee=Decimal("0"), exitoso=False, error_mensaje=str(result)[:200]))
                continue
            if result:
                cotizaciones.append({
                    "courier_codigo": c.codigo, "courier_nombre": c.nombre,
                    "fee": result["fee"], "eta_minutos": result["eta_minutos"],
                    "eta_texto": result["eta_texto"],
                })
                session.add(TCotizacionCourier(
                    papeleria_id=tenant_id, courier_codigo=c.codigo,
                    fee=result["fee"], eta_minutos=result["eta_minutos"], exitoso=True))
        cotizaciones.sort(key=lambda x: x["fee"])
        return cotizaciones

    async def _cotizar_uno(self, session, tenant_id: UUID, courier,
                           origen: dict, destino: dict, peso_kg: Decimal) -> Optional[dict]:
        try:
            if courier.codigo == "FLOTA_INTERNA":
                return await self._cotizar_flota_interna(tenant_id, origen, destino, peso_kg)
            elif courier.codigo == "UBER_DIRECT":
                return await self._cotizar_uber_direct(tenant_id, origen, destino)
            elif courier.codigo == "DIDI":
                return await self._cotizar_didi(tenant_id, origen, destino)
            elif courier.codigo == "INDRIVER":
                return {"fee": 55.00, "eta_minutos": 45, "eta_texto": "45 min"}
            elif courier.codigo == "MOTOTAXI_SEGURO":
                return {"fee": 50.00, "eta_minutos": 40, "eta_texto": "40 min"}
            elif courier.codigo in ("AMAZON_LOGISTICS", "ESTAFETA", "DHL_STANDARD"):
                return await self._cotizar_nacional(courier.codigo, peso_kg)
            return None
        except Exception as e:
            logger.warning("courier.cotizacion_fallida", courier=courier.codigo, error=str(e)[:100])
            raise

    async def _cotizar_flota_interna(self, tenant_id, origen, destino, peso_kg) -> dict:
        base = Decimal("40.00")
        por_kg = Decimal("5.00")
        distancia = self._haversine(origen.get("lat"), origen.get("lng"),
                                    destino.get("lat"), destino.get("lng"))
        fee = base + (peso_kg * por_kg) + (Decimal(str(distancia)) * Decimal("2.00"))
        eta_min = int((distancia * 3) + 20)
        return {"fee": float(round(fee, 2)), "eta_minutos": eta_min,
                "eta_texto": f"{eta_min} min" if eta_min < 60 else f"{eta_min // 60}h {eta_min % 60}m"}

    async def _cotizar_uber_direct(self, tenant_id, origen, destino) -> dict:
        distancia = self._haversine(origen.get("lat"), origen.get("lng"),
                                    destino.get("lat"), destino.get("lng"))
        fee = Decimal("75.00") + (Decimal(str(distancia)) * Decimal("8.00"))
        eta_min = int(distancia * 2.5) + 15
        return {"fee": float(round(fee, 2)), "eta_minutos": eta_min, "eta_texto": f"{eta_min} min"}

    async def _cotizar_didi(self, tenant_id, origen, destino) -> dict:
        distancia = self._haversine(origen.get("lat", 0), origen.get("lng", 0),
                                    destino.get("lat", 0), destino.get("lng", 0))
        fee = Decimal("65.00") + (Decimal(str(distancia)) * Decimal("7.50"))
        eta_min = int(distancia * 3) + 20
        return {"fee": float(round(fee, 2)), "eta_minutos": eta_min, "eta_texto": f"{eta_min} min"}

    async def _cotizar_nacional(self, courier_codigo: str, peso_kg: Decimal) -> dict:
        tarifas = {"AMAZON_LOGISTICS": Decimal("120.00"),
                   "ESTAFETA": Decimal("100.00"),
                   "DHL_STANDARD": Decimal("150.00")}
        base = tarifas.get(courier_codigo, Decimal("100.00"))
        extra_peso = max(Decimal("0"), (peso_kg - Decimal("5")) * Decimal("15.00"))
        return {"fee": float(base + extra_peso), "eta_minutos": 4320, "eta_texto": "3-5 días"}

    @staticmethod
    def _haversine(lat1, lng1, lat2, lng2) -> float:
        import math
        if not all([lat1, lng1, lat2, lng2]):
            return 5.0
        R = 6371
        dlat = math.radians(lat2 - lat1)
        dlng = math.radians(lng2 - lng1)
        a = math.sin(dlat / 2) ** 2 + math.cos(math.radians(lat1)) * math.cos(math.radians(lat2)) * math.sin(dlng / 2) ** 2
        c = 2 * math.asin(math.sqrt(a))
        return round(R * c, 2)

import json, os
from datetime import datetime, timezone
from typing import Optional
from uuid import UUID
import structlog
from aiokafka import AIOKafkaProducer
from src.infrastructure.kafka.schema_validator import validate_event

logger = structlog.get_logger(__name__)
_producer: Optional[AIOKafkaProducer] = None

async def get_producer() -> Optional[AIOKafkaProducer]:
    global _producer
    if _producer is not None: return _producer
    bootstrap = os.getenv("KAFKA_BOOTSTRAP_SERVERS", "")
    if not bootstrap:
        logger.warning("kafka.sin_bootstrap_servers_modo_dev")
        return None
    try:
        _producer = AIOKafkaProducer(
            bootstrap_servers=bootstrap,
            value_serializer=lambda v: json.dumps(v, default=str).encode(),
            key_serializer=lambda k: k.encode() if k else None,
            enable_idempotence=True, acks="all",
            retry_backoff_ms=500, request_timeout_ms=30000,
        )
        await _producer.start()
        logger.info("kafka.producer_iniciado", bootstrap=bootstrap)
        return _producer
    except Exception as e:
        logger.error("kafka.producer_error", error=str(e)[:200])
        _producer = None
        return None

async def close_producer() -> None:
    global _producer
    if _producer:
        try: await _producer.stop()
        except Exception: pass
        _producer = None

async def publish_event(topic: str, key: str, value: dict, *, validate: bool = True) -> bool:
    if validate:
        validate_event(topic, value)
    producer = await get_producer()
    if producer is None:
        logger.warning("kafka.publish_modo_simulado", topic=topic, event_type=value.get("eventType"))
        return False
    try:
        await producer.send_and_wait(topic=topic, key=key, value=value)
        logger.debug("kafka.published", topic=topic, key=key)
        return True
    except Exception as e:
        logger.error("kafka.publish_error", topic=topic, key=key, error=str(e)[:200])
        raise

def build_envelope(event_id: UUID, tenant_id: UUID, event_type: str,
                   payload: dict, trace_id: Optional[str] = None, version: str = "1.0") -> dict:
    return {
        "eventId": str(event_id), "tenantId": str(tenant_id),
        "eventType": event_type,
        "timestamp": datetime.now(timezone.utc).isoformat(),
        "version": version,
        "traceId": trace_id or str(event_id),
        "payload": payload,
    }

import asyncio
from sqlalchemy import text
from src.infrastructure.db.session import WorkerSessionLocal
import structlog

logger = structlog.get_logger(__name__)

async def sweep_reservations():
    while True:
        try:
            async with WorkerSessionLocal() as session:
                # F20: libera a nivel ARTÍCULO (tarticulos.stock_reservado) — consistente
                # con reservar_stock del cart_service. NO toca tlotes.cantidad_reservada
                # (esa es exclusiva de APARADOS).
                await session.execute(text("""
                    WITH items_expirados AS (
                        UPDATE tcarritoitems SET reserva_activa = false
                        WHERE reserva_activa AND fecha_expiracion_reserva < NOW()
                        RETURNING carrito_id, articulo_id, cantidad
                    ), liberacion AS (
                        UPDATE tarticulos a SET stock_reservado =
                            GREATEST(0, a.stock_reservado - sub.total_cantidad)
                        FROM (SELECT articulo_id, SUM(cantidad) total_cantidad
                              FROM items_expirados GROUP BY articulo_id) sub
                        WHERE a.id = sub.articulo_id RETURNING a.id
                    )
                    UPDATE tcarritos c SET estado = 'expirado'
                    WHERE c.estado = 'activo'
                      AND NOT EXISTS (SELECT 1 FROM tcarritoitems i
                                      WHERE i.carrito_id = c.id AND i.reserva_activa);
                """))
                await session.commit()
        except Exception as e:
            logger.debug("reservation_sweeper.skip", error=str(e)[:100])
        await asyncio.sleep(60)

if __name__ == "__main__":
    asyncio.run(sweep_reservations())

import asyncio, json
from datetime import datetime, timezone
from aiokafka import AIOKafkaConsumer
from sqlalchemy import text
from src.infrastructure.db.session import WorkerSessionLocal
from src.core.config import settings

async def consume_dlq():
    """Persiste eventos DLQ en tdlq_eventos (retención 90d vs 30d del topic crudo).
    FIX F9: historial_intentos JSONB con trazabilidad por intento.
    FIX F26: usa settings.KAFKA_BOOTSTRAP_SERVERS (no hardcodea kafka:9092)."""
    consumer = AIOKafkaConsumer("dlq.events",
                                bootstrap_servers=settings.KAFKA_BOOTSTRAP_SERVERS,
                                group_id="dlq_group")
    await consumer.start()
    try:
        async for msg in consumer:
            ev = json.loads(msg.value.decode())
            async with WorkerSessionLocal() as session:
                await session.execute(text("""
                    INSERT INTO tdlq_eventos (id, outbox_evento_id, papeleria_id, event_type,
                        aggregate_type, aggregate_id, payload, error_ultimo,
                        historial_intentos, trace_id, estado, first_failed_at)
                    VALUES (generate_uuid_v7(), :oeid, :tid, :etipo, :agg, :aggid,
                        CAST(:payload AS jsonb), :err,
                        CAST(:hist AS jsonb), :trace, 'pendiente', NOW())
                    ON CONFLICT (outbox_evento_id) DO UPDATE
                    SET error_ultimo = EXCLUDED.error_ultimo,
                        historial_intentos = tdlq_eventos.historial_intentos ||
                            CAST(:nuevo_intento AS jsonb)
                """), {
                    "oeid": ev.get("eventId"), "tid": ev.get("tenantId"),
                    "etipo": ev.get("eventType"), "agg": ev.get("aggregate_type"),
                    "aggid": ev.get("aggregate_id"),
                    "payload": json.dumps(ev.get("payload", {})),
                    "err": ev.get("error"), "trace": ev.get("traceId"),
                    "hist": json.dumps([{"intento": ev.get("intentos"),
                                         "error": ev.get("error"),
                                         "timestamp": datetime.now(timezone.utc).isoformat()}]),
                    "nuevo_intento": json.dumps({"intento": ev.get("intentos"),
                                                 "error": ev.get("error"),
                                                 "timestamp": datetime.now(timezone.utc).isoformat()})})
                await session.commit()
    finally:
        await consumer.stop()

if __name__ == "__main__":
    asyncio.run(consume_dlq())
