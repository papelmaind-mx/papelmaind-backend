DECLARE v_time timestamptz := clock_timestamp(); v_secs bigint; v_msec bigint; v_buf bytea;

async def allocate_stock_fefo(session, tenant_id: UUID, articulo_id: UUID, cantidad: Decimal):
    art = (await session.execute(select(TArticulo).where(TArticulo.id == articulo_id, TArticulo.papeleria_id == tenant_id).with_for_update())).scalar_one_or_none()
    if not art or (art.stock_actual - art.stock_reservado) < cantidad: raise InsufficientStockError(articulo_id=articulo_id)
    lotes = (await session.execute(select(TLote).where(TLote.papeleria_id == tenant_id, TLote.articulo_id == articulo_id, TLote.estado.in_(("disponible","parcial"))).order_by(TLote.fecha_vencimiento.asc()).with_for_update())).scalars().all()
    pendiente = cantidad
    for lote in lotes:
        if pendiente <= 0: break
        disponible = lote.cantidad_actual - lote.cantidad_reservada
        if disponible <= 0: continue
        take = min(disponible, pendiente)
        lote.cantidad_actual -= take
        if lote.cantidad_actual == 0: lote.estado = "agotado"
        session.add(TMovInventario(papeleria_id=tenant_id, articulo_id=articulo_id, lote_id=lote.id, tipo_movimiento="salida", cantidad=take, motivo="Venta"))
        pendiente -= take
    if pendiente > 0: raise InsufficientStockError(articulo_id=articulo_id)
    art.stock_actual -= cantidad
    # F30: Decrementar stock_reservado al liquidar la reserva del carrito
    art.stock_reservado = max(Decimal("0"), art.stock_reservado - cantidad)
