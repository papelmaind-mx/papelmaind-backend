async def test_webhook_tenant_isolation(client, seed_two_tenants):
    tenant_a, tenant_b = seed_two_tenants
    payload = build_meta_webhook(
        phone_number_id=tenant_a.whatsapp_phone_number_id,
        wa_id=tenant_b.cliente.wa_id,
        mensaje="hola",
    )
    resp = await client.post("/api/v1/whatsapp/webhook", json=payload)
    assert resp.status_code == 200
    async with TenantUnitOfWork(tenant_b.id) as session:
        msgs = (await session.execute(select(TMensaje))).scalars().all()
        assert len(msgs) == 0, "F19 roto: escritura cross-tenant"
    async with TenantUnitOfWork(tenant_a.id) as session:
        msgs = (await session.execute(select(TMensaje))).scalars().all()
        assert len(msgs) == 1

async def test_sweeper_consistencia_vw_stock(db, seed_articulo_con_lotes):
    tenant_id = seed_articulo_con_lotes.papeleria_id
    articulo_id = seed_articulo_con_lotes.id
    async with TenantUnitOfWork(tenant_id) as session:
        await reservar_stock(session, tenant_id, articulo_id, Decimal("5"))
    async with WorkerSessionLocal() as session:
        await session.execute(text("UPDATE tcarritoitems SET fecha_expiracion_reserva = NOW() - INTERVAL '1 min' WHERE reserva_activa = true"))
        await session.commit()
    await sweep_reservations_once()
    async with TenantUnitOfWork(tenant_id) as session:

        vw = (await session.execute(text("""
            SELECT disponible_sucursal FROM vw_stock_sucursal
            WHERE papeleria_id = :t AND articulo_id = :a
        """), {"t": str(tenant_id), "a": str(articulo_id)})).scalar()

<anexo id="M" nombre="Glosario de términos de negocio y arquitectura">
<termino nombre="Aparado">Sistema de apartado con enganche (≥30% del total) y plazo (30 días default). RESERVA stock (cantidad_reservada en lotes), NO descuenta. Al liquidar → hard-allocate FEFO. Al vencer → libera + penalización 10% del saldo (mín $10).</termino>
<termino nombre="Caja">Entidad de DINERO (cajas_registradoras). Independiente por caja: fondo, ventas, retiros, cierre. NO confundir con "Estación" (terminal).</termino>
<termino nombre="CFDI">Comprobante Fiscal Digital por Internet (México, SAT 4.0). Timbrado vía PAC (SW Smarter, Facturama, Ecodex, Pocket CFDI).</termino>
<termino nombre="Corte de caja">SESIÓN DE DINERO de UNA caja. UNIQUE(caja_id) WHERE estado='abierto' → una caja = un corte abierto a la vez. |descuadre| &gt; $50 → estado 'observacion'.</termino>
<termino nombre="DLQ (Dead Letter Queue)">Cola de eventos que fallaron 3+ veces. Persistidos en tdlq_eventos con historial_intentos JSONB + retención 90 días. Replay manual o automático.</termino>
<termino nombre="Estación">TERMINAL físico o virtual (POS, WEB, MOVIL, API, KIOSCO, EMPAQUE). Puede tener caja_id nullable (POS → sí; WEB/API → NULL).</termino>
<termino nombre="FEFO (First Expired, First Out)">Algoritmo de despacho de lotes: primero el que vence antes. Lock artículo PRIMERO + lotes después (evita lost-update). Bloquea filas FOR UPDATE.</termino>
<termino nombre="Outbox">Patrón transaccional: cada evento se INSERTA en la tabla outbox_eventos DENTRO de la MISMA transacción ACID del agregado. Workers consumen después. NUNCA usar trigger de BD. F17: flags separados (publicado_kafka / procesado_loyalty).</termino>
<termino nombre="PAC (Proveedor Autorizado de Certificación)">Empresa autorizada por el SAT para timbrar CFDI. PapelMaind usa PAC-FIRST: PAC_PRIMARY → PAC_FALLBACK → DLQ nocturno.</termino>
<termino nombre="Papelería (en DB)">= Tenant. La entidad `papelerias`. Toda tabla con papeleria_id lleva RLS.</termino>
<termino nombre="RLS (Row-Level Security)">Seguridad a nivel de fila de PostgreSQL. PapelMaind usa ENABLE + FORCE + política `tenant_isolation_&lt;tabla&gt;` con USING y WITH CHECK.</termino>
<termino nombre="Tarjeta Consentido">Programa de fidelización. Acumulación 1 peso = 1 punto (P1). Niveles: bronce (0-999) · plata (1k-4,999) · oro (5k-9,999) · platino (10k+). Redención: 100 pts = $5, mínimo 500 pts, tope 50% del ticket.</termino>
<termino nombre="Tenant">Cliente SaaS de PapelMaind. Cada tenant tiene slug → subdominio. IDs UUID v7. RLS aislado.</termino>
<termino nombre="TenantUnitOfWork">Context manager async que setea `app.current_tenant_id` con SET LOCAL transaccional. Obligatorio en tráfico autenticado (API con erp_app).</termino>
<termino nombre="TOCTOU (Time-of-Check to Time-of-Use)">Race condition: entre verificar y actuar, otro proceso cambia el estado. PapelMaind la evita con UNIQUE constraints + ON CONFLICT (idempotencia) en lugar de SELECT previo.</termino>
<termino nombre="UUID v7">UUID timestamp-ordered (RFC 9562). PapelMaind lo usa en TODO (server + POS client_uuid). Genera: `generate_uuid_v7()` en PostgreSQL.</termino>
<termino nombre="WorkerSessionLocal">Session factory para los 9 workers, usando `worker_engine` con rol `erp_worker` (BYPASSRLS). OBLIGATORIO — AsyncSessionLocal en un worker devuelve 0 filas por RLS sin contexto tenant.</termino>
<termino nombre="BYO-LLM (Bring Your Own LLM)">Motor de IA agnóstico que permite conectar o alternar dinámicamente entre diferentes proveedores (Kimi, Qwen, DeepSeek, Claude, OpenAI GPT-4o, modelos locales) mediante API keys o pasarelas.</termino>
<termino nombre="Handoff Inteligente">Sistema de escalamiento automático que detecta cuándo derivar la conversación a un asesor humano basándose en disparadores (reglas de negocio, sentimiento/frustración, cierre B2B).</termino>
<termino nombre="CRM Omnicanal Unificado">Panel de gestión centralizada donde confluyen todas las interacciones de los clientes (WhatsApp, Instagram, Facebook Messenger, Telegram, Email, Web Chat) con contexto de tenant/cliente acoplado al RLS.</termino>
</anexo>
